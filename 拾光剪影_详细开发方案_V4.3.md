### 2.2 @cached装饰器缺陷修复

**文件**: `app/core/cache_system.py`
**位置**: 第569行附近
**问题**: 装饰器每次创建新缓存实例，缓存不共享

#### 当前错误代码

```python
# 第569行附近 - 错误
def cached(ttl: int = 300, key_prefix: str = "", persist: bool = False):
    def decorator(func: Callable) -> Callable:
        cache = HybridCache()  # 每次装饰都创建新实例！
```

#### 修复方案

```python
# 修复后
_GLOBAL_CACHE = HybridCache()  # 模块级全局单例

def cached(ttl: int = 300, key_prefix: str = "", persist: bool = False):
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            # 使用全局缓存
            key = f"{key_prefix}:{func.__name__}:{str(args)}:{str(kwargs)}"
            result = _GLOBAL_CACHE.get(key)
            if result is not None:
                return result
            result = func(*args, **kwargs)
            _GLOBAL_CACHE.set(key, result, ttl=ttl)
            return result
        return wrapper
    return decorator
```

### 2.3 SHA256无盐降级修复

**文件**: `app/core/security.py`
**位置**: 第656-658行
**问题**: 纯SHA256无盐降级，极不安全

#### 当前错误代码

```python
# 第656-658行 - 错误
return hashlib.sha256(password.encode()).hexdigest() == hashed
# 纯SHA256无盐，可被彩虹表攻击
```

#### 修复方案

```python
# 修复后 - 标记为废弃，强制使用bcrypt
def hash_password(self, password: str) -> str:
    """哈希密码"""
    # 优先使用bcrypt
    try:
        import bcrypt
        salt = bcrypt.gensalt(rounds=12)
        hashed = bcrypt.hashpw(password.encode('utf-8'), salt)
        return hashed.decode('utf-8')
    except ImportError:
        # 如果bcrypt不可用，使用argon2
        try:
            import argon2
            ph = argon2.PasswordHasher()
            return ph.hash(password)
        except ImportError:
            # 最后的降级方案 - 使用PBKDF2
            import hashlib, os
            salt = os.urandom(32)
            key = hashlib.pbkdf2_hmac('sha256', password.encode(), salt, 100000)
            return f"$pbkdf2-sha256${salt.hex()}${key.hex()}"
        # 移除不安全的SHA256-only降级
```

```python
                'error_msg': '⚠️ 爆款引擎初始化失败',
                'instance': None
            },
            # ... 其他引擎配置
        }

    def _lazy_load(self, name: str):
        """通用懒加载方法"""
        config = self._lazy_managers.get(name)
        if not config:
            return None

        flag = config['flag']
        if getattr(self, flag, False):
            return config['instance']

        setattr(self, flag, True)
        if not config['available']:
            print(config['error_msg'])
            return None

        try:
            config['instance'] = config['class']()
            print(config['success_msg'])
        except Exception as e:
            print(f"{config['error_msg']}: {e}")

        return config['instance']

    def _ensure_top0_engine(self):
        return self._lazy_load('top0')

    def _ensure_viral_engine(self):
        return self._lazy_load('viral')
```

```python
class LipSyncEngine:
    """嘴型同步引擎"""

    def __init__(self):
        self._audio_processor = None
        self._model = None

    def initialize(self):
        """懒加载初始化"""
        if self._model is not None:
            return

        try:
            # 加载Wav2Lip模型
            import torch
            self._model = torch.load('models/wav2lip.pth')
            self._model.eval()
            self._audio_processor = AudioProcessor()
        except ImportError:
            print("⚠️ 嘴型同步需要安装torch和 librosa")

    def sync(self, audio_path: str, video_path: str, output_path: str) -> Dict:
        """执行嘴型同步"""
        self.initialize()

        # 1. 提取音频特征
        mel_spec = self._extract_mel_spectrogram(audio_path)

        # 2. 生成嘴型参数
        visemes = self._predict_visemes(mel_spec)

        # 3. 应用到视频
        self._apply_visemes_to_video(video_path, visemes, output_path)

        return {
            'success': True,
            'output_path': output_path,
            'viseme_count': len(visemes)
        }

    def _extract_mel_spectrogram(self, audio_path: str) -> np.ndarray:
        """提取梅尔频谱"""
        import librosa
        y, sr = librosa.load(audio_path)
        mel = librosa.feature.melspectrogram(y=y, sr=sr, n_mels=80)
```

```python
# app/core/keyframe_editor.py
from PyQt6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout
from PyQt6.QtCore import Qt, QPointF, pyqtSignal
import numpy as np

class BezierCurveEditor(QWidget):
    """贝塞尔曲线编辑器"""

    # 信号
    curve_changed = pyqtSignal(list)  # 发送曲线控制点列表

    def __init__(self, parent=None):
        super().__init__(parent)
        self._control_points = [
            QPointF(0, 0),
            QPointF(0.25, 0.5),
            QPointF(0.75, 0.5),
            QPointF(1, 1)
        ]
        self._dragging_point = None
        self._setup_ui()

    def _setup_ui(self):
        """设置UI"""
        layout = QVBoxLayout()

        # 控制点编辑器
        self._points_layout = QHBoxLayout()
        for i, point in enumerate(self._control_points):
            x_label = QLabel(f"P{i} X:")
            x_input = QLineEdit(str(point.x()))
            y_label = QLabel(f"Y:")
            y_input = QLineEdit(str(point.y()))
            self._points_layout.addWidget(x_label)
            self._points_layout.addWidget(x_input)
            self._points_layout.addWidget(y_label)
            self._points_layout.addWidget(y_input)

        layout.addLayout(self._points_layout)

        # 预设选择
        presets_layout = QHBoxLayout()
        presets_layout.addWidget(QLabel("预设:"))
        for name, points in self._PRESETS.items():
            btn = QPushButton(name)
            btn.clicked.connect(lambda _, p=points: self._apply_preset(p))
            presets_layout.addWidget(btn)
        layout.addLayout(presets_layout)

        self.setLayout(layout)

    _PRESETS = {
        "线性": [QPointF(0, 0), QPointF(1, 1)],
        "缓入": [QPointF(0, 0), QPointF(0.5, 0), QPointF(1, 1)],
        "缓出": [QPointF(0, 0), QPointF(0.5, 1), QPointF(1, 1)],
        "缓入缓出": [QPointF(0, 0), QPointF(0.25, 0), QPointF(0.75, 1), QPointF(1, 1)],
        "弹性": [QPointF(0, 0), QPointF(0.3, -0.2), QPointF(0.7, 1.2), QPointF(1, 1)],
    }

    def _apply_preset(self, points: list):
        """应用预设"""
        self._control_points = [QPointF(p.x(), p.y()) for p in points]
        self.update()
        self.curve_changed.emit(self._control_points)

    def get_value_at(self, t: float) -> float:
        """获取曲线在t位置的值 (0-1)"""
        # 使用贝塞尔公式计算
        points = self._control_points
        n = len(points) - 1
        value = 0
        for i, p in enumerate(points):
            value += p.y() * self._binomial(n, i) * (1-t)**(n-i) * t**i
        return value
```

```python
        """轨道锁定"""
        if track_type in self._track_widgets:
            widget = self._track_widgets[track_type]
            widget.setLocked(locked)
            for clip in widget.get_clips():
                clip.setEnabled(not locked)

    def _toggle_track_mute(self, track_type: str, muted: bool):
        """轨道静音"""
        if track_type in self._track_widgets:
            widget = self._track_widgets[track_type]
            widget.setMuted(muted)
            for clip in widget.get_clips():
                clip.setAudioMuted(muted)

    def _toggle_track_solo(self, track_type: str, solo: bool):
        """轨道独奏"""
        # 关闭其他轨道
        if solo:
            for t, w in self._track_widgets.items():
                if t != track_type:
                    w.setMuted(True)
                    w.setSolo(False)
        self._track_widgets[track_type].setSolo(solo)

    def _snap_to_guides(self, clip_pos: float) -> float:
        """吸附对齐到辅助线和其他片段"""
        snap_threshold = 0.1  # 秒

        # 吸附到时间线标记
        for marker in self._timeline_markers:
            if abs(clip_pos - marker) < snap_threshold:
                return marker

        # 吸附到其他片段边界
        for widget in self._track_widgets.values():
            for clip in widget.get_clips():
                clip_start = clip.getStartTime()
                clip_end = clip.getEndTime()
                if abs(clip_pos - clip_start) < snap_threshold:
                    return clip_start
                if abs(clip_pos - clip_end) < snap_threshold:
                    return clip_end

        return clip_pos  # 无吸附，返回原位置

    def _apply_ripple_edit(self, clip, delta: float):
        """波纹编辑 - 移动片段后自动调整后续片段"""
        affected_clips = []

        # 找到所有受影响的片段
        track = clip.getTrack()
        clip_end = clip.getEndTime()

        for other_clip in track.get_clips():
            if other_clip.getStartTime() >= clip_end:
                affected_clips.append(other_clip)

        # 移动当前片段
        clip.setStartTime(clip.getStartTime() + delta)

        # 调整后续片段
        for other_clip in affected_clips:
            other_clip.setStartTime(
                other_clip.getStartTime() + delta
            )
```

```python
class InputValidator:
    """增强输入验证"""

    @staticmethod
    def validate_file_path(path: str) -> ValidationResult:
        """验证文件路径"""
        # 检查路径遍历
        if ".." in path or path.startswith("/"):
            return ValidationResult(False, "非法路径")

        # 检查危险字符
        dangerous_chars = ['<', '>', '|', '\0']
        for char in dangerous_chars:
            if char in path:
                return ValidationResult(False, f"路径包含危险字符: {char}")

        # 检查文件大小
        if os.path.exists(path):
            size = os.path.getsize(path)
            if size > 10 * 1024 * 1024 * 1024:  # 10GB
                return ValidationResult(False, "文件过大")

        return ValidationResult(True)

    @staticmethod
    def validate_api_key(key: str, provider: str) -> ValidationResult:
        """验证API Key格式"""
        min_lengths = {
            'openai': 50,
            'anthropic': 50,
            'deepseek': 30,
            'default': 20
        }

        min_len = min_lengths.get(provider, min_lengths['default'])

        if len(key) < min_len:
            return ValidationResult(False, f"API Key长度不足")

        if not re.match(r'^[A-Za-z0-9_-]+$', key):
            return ValidationResult(False, "API Key格式无效")

        return ValidationResult(True)
```
