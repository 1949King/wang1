# 0. 一句话架构
在线  = 纯前向 + 打分 + 落记忆 + 路由         （无梯度，ROCm 稳）
睡眠  = 离线分块全精度反向 + 重放 + 负载均衡  （绕开 4-bit 反向量题）
巩固  = LoRA/专家一等公民，版本化走灰度        （core-merge 是低频大版本）
元认知 = Decision Core 复用，不铸 θ_meta       （只读，O(1)/O(log n)）
奖励  = 只吃 human_label_ledger                （AI 分不进梯度）
专家  = L0 能力头 → L1 翻译模块 → L2 独立专家  （A+B 增长）
理解  = 基座 27B 负责；专家只负责翻译/验证     （不追求通用能力）

import torch, torch.nn as nn

class FrozenReferenceEncoder:
    """不变量 I6：novelty 的尺子，模型改不动。"""
    def __init__(self, path: str, device: str = "cuda"):
        self.device = device
        self.enc = self._load(path).to(device).eval()
        for p in self.enc.parameters():
            p.requires_grad = False

    def _load(self, path: str) -> nn.Module:
        ...

    @torch.no_grad()
    def encode(self, x: torch.Tensor) -> torch.Tensor:
        return self.enc(x).mean(dim=1)

    @torch.no_grad()
    def novelty(self, x: torch.Tensor, bank: torch.Tensor) -> float:
        if bank.shape[0] == 0:
            return 1.0
        z = self.encode(x).unsqueeze(0)
        sim = torch.cosine_similarity(z, bank, dim=-1)
        return float(1.0 - sim.max())

# sleep_subsystem/online_collector.py
import torch, uuid
from collections import deque
from typing import Callable

class OnlineCollector:
    """不变量 I1：只前向。不变量 I13：基座 4-bit 推理。"""
    def __init__(self, base_model, encoder, bank,
                 max_episodes: int = 100_000, device: str = "cuda"):
        self.base = base_model.to(device).eval()
        for p in self.base.parameters():
            p.requires_grad = False
        self.encoder = encoder
        self.bank = bank.to(device)
        self.eps: deque = deque(maxlen=max_episodes)
        self.device = device

    @torch.no_grad()
    def step(self, input_ids, decision_id, model_ver, reward_proxy_fn) -> Episode:
        input_ids = input_ids.to(self.device)
        output_ids = self.base.generate(input_ids, max_new_tokens=512, do_sample=False)
        r = reward_proxy_fn(input_ids, output_ids)
        nov = self.encoder.novelty(input_ids, self.bank)
        ep = Episode(str(uuid.uuid4()), input_ids.cpu(), output_ids.cpu(),
                     r, nov, decision_id, model_ver)
        self.eps.append(ep)
        return ep

    def drain(self, model_ver: str, limit: int = 1000) -> list:
        return [e for e in self.eps if e.model_ver == model_ver][:limit]

# sleep_subsystem/router.py
import torch, torch.nn as nn

class LoadBalancedRouter(nn.Module):
    """不变量 I8/I9/I10。"""
    def __init__(self, dim: int, max_experts: int = 64):
        super().__init__()
        self.dim = dim
        self.max_experts = max_experts
        self.gate = nn.Linear(dim, max_experts)
        self.n_active = 0
        with torch.no_grad():
            self.gate.weight.data.mul_(1e-2)
            self.gate.bias.data.zero_()

    def forward(self, x):
        pooled = x.mean(dim=1)
        logits = self.gate(pooled)[:, : self.n_active]
        return torch.softmax(logits, dim=-1)

    @torch.no_grad()
    def grow_one(self) -> int:
        if self.n_active >= self.max_experts:
            raise RuntimeError("max_experts reached")
        idx = self.n_active
        self.gate.weight.data[idx] = torch.randn_like(self.gate.weight[0]) * 1e-2
        self.gate.bias.data[idx] = 0.0
        self.n_active += 1
        return idx

    def load_balance_loss(self, probs):
        if self.n_active <= 1:
            return torch.tensor(0.0, device=probs.device)
        usage = probs.mean(dim=0)
        target = torch.ones_like(usage) / self.n_active
        return (usage - target).pow(2).sum()

    @torch.no_grad()
    def prune(self, usage_ema, threshold: float = 1e-3):
        for i in range(self.n_active):
            if usage_ema[i] < threshold:
                self._retire(i)

    def _retire(self, idx: int):

import torch, random
from typing import List

class FPBlockLoader:
    """不变量 I2/I13：4-bit 基座不直接反向，分块全精度。"""
    def __init__(self, ckpt_path: str, block_size: int = 2, device: str = "cuda"):
        self.ckpt_path = ckpt_path
        self.block_size = block_size
        self.device = device
        self._blocks = self._index_blocks()

    def _index_blocks(self): ...
    def chunked_blocks(self):
        for i in range(len(self._blocks)):
            yield self._load_block(i, dtype=torch.bfloat16)
    def _load_block(self, idx, dtype): ...
    def snapshot(self) -> str: ...
    def requantize(self, block): ...

class SleepConsolidator:
    """不变量 I4/I5/I7/I11。"""
    def __init__(self, fp_loader, lora, experts, router,
                 ledger, mvr, dispatcher, feature_bus,
                 novelty_threshold: float = 0.7, ewc_lambda: float = 1e-3):
        self.fp_loader = fp_loader
        self.lora = lora
        self.experts = experts
        self.router = router
        self.ledger = ledger
        self.mvr = mvr
        self.dispatcher = dispatcher
        self.feature_bus = feature_bus
        self.novelty_threshold = novelty_threshold
        self.ewc_lambda = ewc_lambda
        self.optim = torch.optim.AdamW(
            list(lora.parameters()) + list(router.parameters()), lr=1e-4)
        self.usage_ema = torch.zeros(router.max_experts)

    def sleep_cycle(self, episodes: List[Episode]) -> ConsolidationResult:
        model_ver_in = self.mvr.current()
        labels = self.ledger.get_labels(model_ver_in, limit=1000)
        if not labels:
            return ConsolidationResult("", model_ver_in, "none", 0.0,
                                       False, "no_human_labels")

        replay = random.sample(episodes, min(500, len(episodes)))
        batch = self._merge(labels, replay)

        for blk in self.fp_loader.chunked_blocks():
            blk.train()
            with torch.enable_grad():
                loss = self._compute_loss(blk, batch)
                loss.backward()
            self.optim.step()
            self.optim.zero_grad()
            blk.eval()
            self.fp_loader.requantize(blk)
            del blk
            torch.cuda.empty_cache()

        self._propose_expansions(episodes)
        self.router.prune(self.usage_ema)
        consistency = self._shadow_consistency(batch)

        if consistency >= 0.95 and self._should_bump_core():
            new_ver = self.mvr.bump("core_merge", self.fp_loader.snapshot())
        else:
            new_ver = model_ver_in

        return ConsolidationResult("", new_ver, "lora+experts",
                                   consistency, consistency >= 0.9, "ok")

    def _compute_loss(self, blk, batch):
        loss_task = self._task_loss(blk, batch)
        loss_replay = self._replay_loss(blk, batch)
        loss_ewc = self.ewc_lambda * self._ewc_penalty()
        probs = self.router(self._router_input(batch))
        loss_balance = 0.01 * self.router.load_balance_loss(probs)
        return loss_task + loss_replay + loss_ewc + loss_balance

    def _task_loss(self, blk, batch): ...
    def _replay_loss(self, blk, batch): ...
    def _ewc_penalty(self): ...
    def _router_input(self, batch): ...
    def _merge(self, labels, replay): ...
    def _shadow_consistency(self, batch) -> float: ...
    def _should_bump_core(self) -> bool: ...

    def _propose_expansions(self, episodes):
        for ep in episodes:
            if ep.novelty_score < self.novelty_threshold:
                continue
            self.dispatcher.submit(DecisionOutput(
                type="expand_expert",
                decision_id=ep.decision_id,
                payload={"episode_id": ep.episode_id,
                         "novelty": ep.novelty_score,
                         "model_ver": ep.model_ver},
            ))

from dataclasses import dataclass
from typing import Literal

@dataclass
class ConsolidationRequest:
    request_id: str
    model_ver_in: str
    episodes_digest: str
    labels_digest: str
    target: Literal["lora", "expert", "core", "hetero_expert"]
    expert_type: str = "dense_transformer"
    estimated_cost: float = 0.0
    decision_id: str = ""

class ConsolidationDispatcher:
    def __init__(self, policy, quota, trace):
        self.policy = policy
        self.quota = quota
        self.trace = trace

    def submit(self, req: ConsolidationRequest) -> str:
        if not self.quota.acquire(req.estimated_cost):
            return "rejected:quota"
        if req.target == "core" and not self.policy.approve_core_merge(req):
            return "rejected:policy"
        self.trace.record(req)
        return self._dispatch(req)

## 4.2 model_ver 发布流程
1. 触发条件
   - 累积 N 次 LoRA/专家巩固
   - 影子一致率 ≥ 0.95 连续 M 轮
   - 预算充足（PI λ 低）

2. 快照
   - 全精度 checkpoint 落盘
   - 记录 Fisher / usage_ema / 路由分布

3. 构建新 core
   - 合并 LoRA 到全精度 core
   - 重新量化 NF4
   - 生成新 model_ver

4. 影子阶段
   - 旧 core 与新 core 并行推理
   - 计算影子一致率、perplexity delta、分位数漂移

5. 灰度
   - 流量 1% → 10% → 50% → 100%
   - 监控 Feature Bus 命中率 / Score Engine 分位

6. 切换
   - bump model_ver
   - Feature Bus 自动隔离
   - 标定分位数 / PI S / H_t 冷启动继承

7. 回滚（任一阶段失败）
   - 恢复快照
   - model_ver 回退
   - trace 记录原因

## 4.3 Feature Bus 版本隔离
# decision_core/feature_bus.py
import hashlib, json

def canonical_input_digest(payload: dict, model_ver: str,
                           modality_types: list = None) -> str:
    """第 8 轮不变量：model_ver 进 digest。"""
    h = hashlib.sha256()
    h.update(json.dumps(payload, sort_keys=True).encode())
    h.update(model_ver.encode())
    if modality_types:
        h.update(json.dumps(sorted(modality_types)).encode())
    return h.hexdigest()

class FeatureBus:
    def get(self, payload, model_ver, modality_types=None):
        return self._cache.get(canonical_input_digest(payload, model_ver, modality_types))
    def put(self, payload, model_ver, value, modality_types=None):
        self._cache[canonical_input_digest(payload, model_ver, modality_types)] = value

## 4.4 睡眠桥接
# decision_core/sleep_bridge.py
class SleepBridge:
    def __init__(self, collector, consolidator, dispatcher, auction, mvr, trace):
        self.collector = collector
        self.consolidator = consolidator
        self.dispatcher = dispatcher
        self.auction = auction
        self.mvr = mvr
        self.trace = trace
        self._running = False

    def tick(self):
        if self._running:
            return
        self._running = True
        try:
            model_ver = self.mvr.current()
            episodes = self.collector.drain(model_ver, limit=2000)
            if not episodes:
                return
            req = ConsolidationRequest(
                request_id=str(uuid.uuid4()),
                model_ver_in=model_ver,
                episodes_digest=digest(episodes),
                labels_digest=digest(self.consolidator.ledger),
                target="lora+experts",
                estimated_cost=self._estimate_cost(episodes),
                decision_id=episodes[0].decision_id,
            )
            status = self.dispatcher.submit(req)
            if status.startswith("rejected"):
                self.trace.record({"event": "consolidation_rejected", "reason": status})
                return
            result = self.consolidator.sleep_cycle(episodes)
            self.trace.record({"event": "consolidation_done", "result": result})
        finally:
            self._running = False

# 5. EWC 的定位
EWC 在低秩重参数化 + 量化空间下 Fisher 估计不可靠，且在大规模持续学习里早被证明不如 replay + 层分离本身有效。本系统已有 replay 和 core/plastic 分离两件更强的武器，EWC 仅作睡眠期弱正则，ewc_lambda=1e-3，不作抗遗忘支柱。

from typing import Iterator, Dict
import torch, torch.nn as nn

class HeteroBlockLoader(ABC):
    @abstractmethod
    def chunked_blocks(self) -> Iterator[nn.Module]: ...
    @abstractmethod
    def snapshot(self) -> str: ...
    @abstractmethod
    def requantize(self, block: nn.Module) -> None: ...
    @property
    @abstractmethod
    def precision_map(self) -> Dict[str, str]: ...

class FPBlockLoader(HeteroBlockLoader):
    def __init__(self, ckpt_path, block_size=2, device="cuda",
                 target_dtype=torch.bfloat16):
        ...
    def precision_map(self):
        return {
            "layers.*.self_attn": "nf4",
            "layers.*.mlp": "nf4",
            "embed_tokens": "bf16",
            "norm": "bf16",
            "lm_head": "bf16",

    def chunked_blocks(self):

class HeteroOnlineCollector(OnlineCollector):
    @torch.no_grad()
    def step_multimodal(self, input_ids, modality_refs, expert_type,
                        decision_id, model_ver, reward_proxy_fn, novelty_fn):
        if expert_type == "dit":
            z_cond = self.base.encode_condition(input_ids, modality_refs)
            output = self._run_dit_expert(z_cond, expert_type)
        elif expert_type == "tts":
            output = self._run_tts_expert(input_ids)
        else:
            output = self.base.generate(input_ids)

        r_components = reward_proxy_fn(input_ids, output, modality_refs, expert_type)
        r = self._aggregate_reward(r_components)
        nov_components = novelty_fn(input_ids, modality_refs, expert_type)
        nov = nov_components["joint_novelty"]

        ep = HeteroEpisode(
            episode_id=str(uuid.uuid4()),
            input_ids=input_ids.cpu(),
            output_ids=output.cpu() if isinstance(output, torch.Tensor) else output,
            reward_proxy=r, novelty_score=nov,
            decision_id=decision_id, model_ver=model_ver,
            modality_refs=modality_refs, expert_type=expert_type,
            reward_components=r_components,
            novelty_components=nov_components)
        self.eps.append(ep)
        return ep

    def _aggregate_reward(self, components):
        if components.get("human_label") is not None:
            return components["human_label"]
        weights = {"clip_score": 0.3, "optical_flow_smoothness": 0.2,
                   "temporal_consistency": 0.2, "schema_valid": 0.15,
                   "resample_consistency": 0.15}
        return sum(weights.get(k, 0) * v for k, v in components.items()
                   if k in weights and isinstance(v, (int, float)))

## 12.4 跨模态 Novelty
class CrossModalNovelty:
    """不变量 I6/I18。"""
    def __init__(self, encoder, text_bank, image_bank, audio_bank, joint_bank):
        self.enc = encoder
        self.banks = {"text": text_bank, "image": image_bank,
                      "audio": audio_bank, "joint": joint_bank}

    @torch.no_grad()
    def compute(self, input_ids, modality_refs, expert_type) -> dict:
        components = {}
        if input_ids is not None:
            z_text = self.enc.encode_multimodal(text=input_ids)
            components["text_novelty"] = self._cosine_novelty(z_text, self.banks["text"])
        for ref in modality_refs:
            if ref.modality == "image":
                z = self.enc.encode_multimodal(image=ref.path)
                components["image_novelty"] = self._cosine_novelty(z, self.banks["image"])
            elif ref.modality == "video":
                z = self.enc.encode_multimodal(video=ref.path)
                components["video_novelty"] = self._cosine_novelty(z, self.banks["joint"])
            elif ref.modality == "audio":
                z = self.enc.encode_multimodal(audio=ref.path)
                components["audio_novelty"] = self._cosine_novelty(z, self.banks["audio"])
        z_joint = self.enc.encode_multimodal(
            text=input_ids, **{r.modality: r.path for r in modality_refs})
        components["joint_novelty"] = self._cosine_novelty(z_joint, self.banks["joint"])
        return components

    def _cosine_novelty(self, z, bank):
        if bank.shape[0] == 0:
            return 1.0
        sim = torch.cosine_similarity(z.unsqueeze(0), bank, dim=-1)
        return float(1.0 - sim.max())

class AsyncGenerationQueue:
    """不变量 I19。视频/图像生成不阻塞在线文本推理。"""
    def __init__(self, max_workers=2, vram_budget_gb=8.0):
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
        self.vram_budget = vram_budget_gb
        self.pending = asyncio.Queue()

    async def enqueue(self, task: dict) -> str:
        task_id = str(uuid.uuid4())
        await self.pending.put((task_id, task))
        return task_id

    async def worker(self):
        while True:
            task_id, task = await self.pending.get()
            try:
                if not self._check_vram(task["expert_type"]):
                    await self.pending.put((task_id, task))
                    await asyncio.sleep(1)
                    continue
                loop = asyncio.get_event_loop()
                result = await loop.run_in_executor(self.executor, self._execute, task)
                await self._store_result(task_id, result)
            except Exception as e:
                self.trace.record({"event": "async_gen_failed",
                                   "task_id": task_id, "error": str(e)})

    def _execute(self, task):
        loader = get_loader(task["expert_type"], **task["loader_kwargs"])
        for blk in loader.chunked_blocks():
            result = blk.generate(task["inputs"])
        loader.requantize(blk)
        return result

class HeteroSleepConsolidator(SleepConsolidator):
    def __init__(self, loader_registry, **kwargs):
        super().__init__(**kwargs)
        self.loaders = loader_registry

    def sleep_cycle_hetero(self, episodes):
        model_ver_in = self.mvr.current()
        groups = {}
        for ep in episodes:
            groups.setdefault(ep.expert_type, []).append(ep)

        for expert_type, eps in groups.items():
            loader = self.loaders[expert_type]
            labels = self.ledger.get_labels(model_ver_in, expert_type, limit=1000)
            if not labels:
                continue
            # 不变量 I20：串行执行
            for blk in loader.chunked_blocks():
                blk.train()
                with torch.enable_grad():
                    loss = self._compute_hetero_loss(blk, eps, labels, expert_type)
                    loss.backward()
                self.optim.step()
                self.optim.zero_grad()
                blk.eval()
                loader.requantize(blk)
                del blk
                torch.cuda.empty_cache()
            self._propose_expansions(eps, expert_type)
